# Sculpt toolbox: registry addendum (interface names only)
# Same source and format as the command-name registry: tool name, then the
# form fields in form order as name:type=start  # UI label.
# Choice fields are integers in the order the spec lists the choices (0 = first;
# checked against the two documented Normal Mode defaults, Flatten and Spin).
# Percent values are 0..1. Distances in metres. REA-assisted: no.
# Activation: tool.set <name> on

Fold => sculpt.fold
    paint:integer=0  # Sculpt Mode
    omod:integer=0  # Offset Mode
    offset:distance=0.1  # Offset Distance
    oval:percent=0.25  # Offset Amount
    sval:percent=0.5  # Smooth Amount
    smod:integer=0  # Smooth Mode
    strokeStep:percent=0.05  # Interpolation Step
    strokeContinuous:boolean=0  # Continuous Stroke
    curveConstraint:integer=0  # Curve Constraint

Clay => sculpt.clay
    oval:percent=0.5  # Offset Amount
    level:percent=0.1  # Level
    sval:percent=0.5  # Smooth Amount
    smod:integer=0  # Smooth Mode
    nmod:integer=2  # Normal Mode
    dmod:integer=1  # Density Mode
    strokeStep:percent=0.05  # Interpolation Step
    strokeContinuous:boolean=0  # Continuous Stroke
    curveConstraint:integer=0  # Curve Constraint
